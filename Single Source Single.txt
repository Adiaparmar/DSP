#include <stdio.h>
#include <conio.h>

#define infinity 999

void dij(int n, int v, int cost[10][10], int dis[10]);

void main()
{
    int n, v, dest;
    int i, j;
    int cost[10][10], dis[10];

    clrscr();

    printf("Enter number of nodes: ");
    scanf("%d", &n);

    printf("\nEnter cost matrix:\n");

    for(i = 1; i <= n; i++)
    {
        for(j = 1; j <= n; j++)
        {
            scanf("%d", &cost[i][j]);
        }
    }

    /* Replace 0 with infinity */
    for(i = 1; i <= n; i++)
    {
        for(j = 1; j <= n; j++)
        {
            if(cost[i][j] == 0)
                cost[i][j] = infinity;
        }
    }

    printf("Enter source vertex: ");
    scanf("%d", &v);

    printf("Enter destination vertex: ");
    scanf("%d", &dest);

    dij(n, v, cost, dis);

    printf("\nShortest distance from %d to %d = %d\n", v, dest, dis[dest]);

    getch();
}


/* Dijkstra Algorithm */
void dij(int n, int v, int cost[10][10], int dis[10])
{
    int i, u, count, w;
    int flag[10];
    int min;

    for(i = 1; i <= n; i++)
    {
        flag[i] = 0;
        dis[i] = cost[v][i];
    }

    flag[v] = 1;
    count = 2;

    while(count <= n)
    {
        min = infinity;

        for(w = 1; w <= n; w++)
        {
            if(dis[w] < min && !flag[w])
            {
                min = dis[w];
                u = w;
            }
        }

        flag[u] = 1;
        count++;

        for(w = 1; w <= n; w++)
        {
            if((dis[u] + cost[u][w] < dis[w]) && !flag[w])
            {
                dis[w] = dis[u] + cost[u][w];
            }
        }
    }
}
