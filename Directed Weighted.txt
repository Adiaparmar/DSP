#include <stdio.h>
#include <conio.h>

#define N 100

/* Graph represented using adjacency matrix */
int Graph[N][N];

/* u = source vertex, v = destination vertex, w = weight */
int vertices, edges;
int u, v, w;
int i, j;

/* Function to input the graph */
void InputGraph()
{
    printf("Enter number of vertices and edges:\n");
    scanf("%d %d", &vertices, &edges);

    /* Initialize adjacency matrix */
    for(i = 1; i <= vertices; i++)
    {
        for(j = 1; j <= vertices; j++)
        {
            Graph[i][j] = 0;
        }
    }

    /* Input edges with weights */
    printf("Enter edges (u v weight):\n");

    for(i = 1; i <= edges; i++)
    {
        printf("Edge-%d: ", i);
        scanf("%d %d %d", &u, &v, &w);

        /* Directed edge from u to v with weight w */
        Graph[u][v] = w;
    }
}

/* Function to print adjacency matrix */
void PrintGraph()
{
    printf("\nAdjacency Matrix of Directed Weighted Graph:\n");

    for(i = 1; i <= vertices; i++)
    {
        for(j = 1; j <= vertices; j++)
        {
            printf("%d\t", Graph[i][j]);
        }
        printf("\n");
    }
}

/* Main function */
void main()
{
    clrscr();

    printf("Directed Weighted Graph\n");
    printf("=======================\n\n");

    InputGraph();
    PrintGraph();

    getch();
}
