clc;

numTx = 2;
numRx = 2;
numSymbols = 1000;
SNR_dB = 20;

% Generate random data symbols
dataSymbols = randi([0 1],numSymbols,numTx);

% Modulation
modulatedSymbols = pskmod(dataSymbols,2);

% MIMO Channel
H = 1/sqrt(2) * ...
    (randn(numRx,numTx) + 1i*randn(numRx,numTx));

% Add noise
snr_linear = 10^(SNR_dB/10);
noisePower = 1/snr_linear;

noise = sqrt(noisePower/2) * ...
    (randn(numSymbols,numRx) + ...
     1i*randn(numSymbols,numRx));

receivedSymbols = modulatedSymbols * H' + noise;

detectedSymbols = pskdemod(receivedSymbols,2);

BER = sum(sum(dataSymbols ~= detectedSymbols)) / ...
    (numSymbols*numTx);

disp(['Bit Error Rate (BER): ' num2str(BER)]);
